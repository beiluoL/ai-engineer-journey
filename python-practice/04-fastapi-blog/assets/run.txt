# 边界情况：校验失败、删除、查不到 —— 这些才是一个接口真正的「设计」

$ curl -s -X POST http://127.0.0.1:8130/posts -H 'Content-Type: application/json' -d '{"title":"","content":"正文"}'
HTTP 422
{
  "detail": [
    {
      "type": "string_too_short",
      "loc": ["body", "title"],
      "msg": "String should have at least 1 character",
      "input": "",
      "ctx": {
        "min_length": 1
      }
    }
  ]
}

$ curl -si -X DELETE http://127.0.0.1:8130/posts/2
HTTP/1.1 204 No Content
date: Wed, 30 Sep 2026 06:30:42 GMT
server: uvicorn


# 删掉之后再查，必须是 404 —— 而不是静默返回空对象

$ curl -s http://127.0.0.1:8130/posts/2
HTTP 404
{
  "detail": "没有 id=2 的文章"
}

$ curl -s -X PUT http://127.0.0.1:8130/posts/999 -H 'Content-Type: application/json' -d '{"title":"x"}'
HTTP 404
{
  "detail": "没有 id=999 的文章"
}
