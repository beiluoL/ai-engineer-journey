八、数据库设计

核心：

6张表。

1. 用户表

user

CREATE TABLE user
(
 id bigint,

 username varchar(50),

 password varchar(255),

 target_job varchar(100),

 created_time datetime
);

2. 面试记录表

interview_session

CREATE TABLE interview_session
(
 id bigint,

 user_id bigint,

 job varchar(100),

 status varchar(20),

 score int,

 created_time datetime
);

3. 面试问题表

question

CREATE TABLE question
(
 id bigint,

 session_id bigint,

 content text,

 category varchar(50)
);

4. 用户回答表

answer

CREATE TABLE answer
(
 id bigint,

 question_id bigint,

 content text,

 score int,

 feedback text
);

5. 知识薄弱点

weak_point

CREATE TABLE weak_point
(
 id bigint,

 user_id bigint,

 knowledge varchar(100),

 level int
);

6. 学习计划

learning_plan

CREATE TABLE learning_plan
(
 id bigint,

 user_id bigint,

 content text,

 status varchar(20)
);
